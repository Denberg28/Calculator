cmake_minimum_required(VERSION 3.20)
file(STRINGS "VERSION" CALCULATOR_VERSION LIMIT_COUNT 1)
project(Calculator VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W4 /WX /permissive- /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()
add_library(calculator_core src/calculator.cpp)
target_include_directories(calculator_core PUBLIC src)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE calculator_core)
add_test(NAME arithmetic_and_state COMMAND core_tests)
if(WIN32)
  enable_language(RC)
  add_executable(Calculator WIN32 src/winmain.cpp resources/app.rc)
  target_compile_definitions(Calculator PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00 CALCULATOR_VERSION="${CALCULATOR_VERSION}")
  target_link_libraries(Calculator PRIVATE calculator_core user32 gdi32)
  target_include_directories(Calculator PRIVATE resources)
  if(MSVC)
    target_link_options(Calculator PRIVATE /MANIFEST:NO /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
  endif()
  add_test(NAME windows_ui COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_SOURCE_DIR}/tests/windows-ui.ps1" -Exe "$<TARGET_FILE:Calculator>")
  set_tests_properties(windows_ui PROPERTIES TIMEOUT 45)
endif()
